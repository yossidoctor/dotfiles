# a fixed wait then a check is a poll however short the label — ScheduleWakeup
# resumes the session instead, with no process held open
deny	sleep 25; gh pr checks 143

# under ten seconds the wait is part of one operation: reaping a killed process,
# settling a connect. The check belongs in the same command.
allow	sleep 5

# loop closed (done) before an unrelated trailing sleep — sleep is foreground but
# not a poll loop; a trailing sub-10s sleep must pass
allow	for x in a b; do echo $x; done; sleep 5
# same shape but a trailing sleep at or above the bar -> caught by the fixed-sleep
# check, not the loop check
deny	for x in a b; do echo $x; done; sleep 60
# nested loop whose inner `done` closes first — sleep is inside the outer loop,
# so stripping from the LAST done keeps it -> real poll loop, must deny
deny	while true; do for i in 1 2; do echo $i; done; sleep 5; done

# A commit message naming a follow stream is a quoted literal, not a command --
# the shape normalizer drops it before the match, so the commit stays on the
# main thread where its exit status is read.

allow	git add x && git commit -m "watch this"

# A real follow stream backgrounds on its own merits, whatever command precedes
# it: the follow flag is what the rewrite keys on, not the leading binary.
bg_forced	git log -p | tail -f

allow	kubectl logs my-pod

# An agent has neither ScheduleWakeup nor Monitor, and a backgrounded sleep delays
# its next turn by nothing, so a held foreground sleep is its only wait. The
# payload carries no caller identity, so `description: poll wait` is the opt-out.
poll_allow	sleep 300
poll_allow	sleep 5m
poll_allow	sleep 45s
# the Bash default timeout is 2 minutes, so a longer sleep returns short unless
# the call carries its own: the hook sets it from the sleep's own duration
poll_timeout	sleep 300	310000
poll_timeout	sleep 5m	310000
poll_timeout	sleep 45s	55000
# the label buys one bare sleep and no workaround built around one
poll_deny	sleep 300 && curl localhost
poll_deny	sleep 300; gh pr checks 1
poll_deny	while true; do sleep 300; done
# unlabelled sleeps are untouched by the carve-out
deny	sleep 300

# Heredoc bodies are stripped before matching, so a command quoted inside one
# is data and never a verdict. That gives up denying a real foreground `sleep`
# written into a heredoc -- the trade that buys a silent rewrite a false-
# positive rate low enough to be trusted, and it retires the unbalanced-quote
# desync an apostrophe in a body used to cause.
allow	bash <<'EOF'\necho "it's fine"\nsleep 60\nEOF
allow	bash <<'EOF'\necho don't\nsleep 60\nEOF
# A heredoc body quoting a follow stream: data, not a command.
allow	git commit -F - <<'EOF'\nrewrites tail -f and gh run watch\nEOF
