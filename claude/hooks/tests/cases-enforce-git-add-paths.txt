# cases for enforce-git-add-paths.sh — expect<TAB>command<TAB>cwd
# Block a `git add` that stages by sweep rather than by path, and `git commit -a`
# which skips the index entirely. Every tree here has other sessions in it, so a
# sweep stages work this session did not write.

# --- denied: the sweep flags ---
deny	git add -A	%HOME%/dotfiles
deny	git add --all	%HOME%/dotfiles
deny	git add -u	%HOME%/dotfiles
deny	git add --update	%HOME%/dotfiles
deny	git add --no-ignore-removal	%HOME%/dotfiles
deny	git add -A .	%HOME%/dotfiles
deny	git add -Av	%HOME%/dotfiles

# --- denied: directory-wide operands ---
deny	git add .	%HOME%/dotfiles
deny	git add ./	%HOME%/dotfiles
deny	git add ..	%HOME%/dotfiles
deny	git add ../	%HOME%/dotfiles
deny	git add -v .	%HOME%/dotfiles
deny	git add :/	%HOME%/dotfiles
deny	git add :(top)	%HOME%/dotfiles

# --- denied: with -C, and mid-sequence ---
deny	git -C ~/otherrepo add -A	%HOME%
deny	git -C %HOME%/dotfiles add .	%HOME%
allow	cd ~/dotfiles && git add README.md	%HOME%
deny	cd ~/dotfiles && git add -A	%HOME%

# --- denied: commit -a bypasses the index ---
deny	git commit -a -m "x"	%HOME%/dotfiles
deny	git commit --all -m "x"	%HOME%/dotfiles
deny	git commit -am "x"	%HOME%/dotfiles
deny	git -C ~/otherrepo commit -a	%HOME%

# --- allowed: named paths, the intended form ---
allow	git add README.md	%HOME%/dotfiles
allow	git add claude/CLAUDE.md claude/settings.json	%HOME%/dotfiles
allow	git add docs/claude/instructing-claude.md	%HOME%/dotfiles
allow	git -C ~/otherrepo add some/nested/file.md	%HOME%
allow	git add path/with.a.dot/file.md	%HOME%/dotfiles
allow	git add .gitignore .editorconfig	%HOME%/dotfiles
allow	git add ./README.md ../other/file.md	%HOME%/dotfiles
allow	git add src/	%HOME%/dotfiles

# --- allowed: hunk-wise and intent-to-add are ownership by construction ---
allow	git add -p	%HOME%/dotfiles
allow	git add --patch	%HOME%/dotfiles
allow	git add -N newfile.md	%HOME%/dotfiles
allow	git add --intent-to-add newfile.md	%HOME%/dotfiles

# --- allowed: a commit without -a, the intended form ---
allow	git commit -m "x"	%HOME%/dotfiles
allow	git commit -q -F -	%HOME%/dotfiles
allow	git commit --amend --no-edit	%HOME%/dotfiles

# --- allowed: other git verbs the gate must not catch ---
allow	git status --short	%HOME%/dotfiles
allow	git diff --cached --stat	%HOME%/dotfiles
allow	git add-ignore	%HOME%/dotfiles
allow	git log --all --oneline	%HOME%/dotfiles
